# ==============================================
# 一、全局通用预处理（最高优先级）
# ==============================================
# 分数去除多余空格

# 全大写英文统一转为标准大小写（抹平大小写差异）
^([A-Z ]+)$=\L$1

# ==============================================
# 二、数值、符号、货币、百分比（原样输出）
# ==============================================
# 纯数字（含千分位）
^\d+(,\d+)*$=$0
# 小数
^\d+\.\d+$=$0
# 百分比
^\d+%$=$0
# 倍数 x开头
^x\d+$=$0
# 美元价格
^\$\d+(\.\d+)?$=$0
# 特殊符号组
^[+_%!:_]+$=$0
# 2~4位纯数字
# 分钟简写格式 xm → x分钟
^(\d+)m$=$1分钟

# ==============================================
# 三、中文语句精简/润色规则
# ==============================================
# 句式重构：建造XX来生产XX
^(.*)建造(.*)来生产(.*)。$=$1建$2，产出$3。
# 句式重构：可以使用XX来XX
^(.*)可以使用(.*)来(.*)。$=$1$2可$3。
# 句式重构：将/会让你XX
^(.*)将让你(.*)。$=$1可$2。
^(.*)会让你(.*)。$=$1可$2。

# 冗余助词替换
(以及|并且)=$、

# 地点状语精简
^(.*)在(.*)中(.*)。$=$1$2内$3。
^(.*)在(.*)里(.*)。$=$1$2内$3。

# 时间单位简写

# 疑问句式精简
^(.*)对(.*)有疑问(.*)。$=$1$2有疑问$3。
